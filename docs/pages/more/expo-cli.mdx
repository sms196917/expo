import React, { useState } from 'react';
import { StyleSheet, View, TextInput, Button, FlatList, Text, TouchableOpacity } from 'react-native';
import { Checkbox } from 'react-native-paper';
import PushNotification from 'react-native-push-notification';
import { Calendar } from 'react-native-calendars';
import ColorPicker from 'react-native-color-picker';

PushNotification.configure({
    onRegister: function (token) {
        console.log("TOKEN:", token);
    },
    onNotification: function (notification) {
        console.log("NOTIFICATION:", notification);
    },
    allowWhileIdle: true,
});

const App = () => {
    const [tasks, setTasks] = useState([]);
    const [taskInput, setTaskInput] = useState('');
    const [noteInput, setNoteInput] = useState('');
    const [searchInput, setSearchInput] = useState('');
    const [selectedDate, setSelectedDate] = useState('');
    const [selectedColor, setSelectedColor] = useState('#ffffff');

    const addTask = () => {
        if (taskInput.trim() && selectedDate) {
            const newTask = {
                text: taskInput.trim(),
                completed: false,
                note: noteInput.trim(),
                date: selectedDate,
                color: selectedColor,
            };
            setTasks([...tasks, newTask]);
            scheduleNotification(newTask);
            clearInputs();
        }
    };

    const scheduleNotification = (task) => {
        const date = new Date(task.date);
        PushNotification.localNotificationSchedule({
            id: task.text,
            message: `予定日です: ${task.text}\nメモ: ${task.note}`,
            date: date,
            allowWhileIdle: true,
        });
    };

    const clearInputs = () => {
        setTaskInput('');
        setNoteInput('');
        setSelectedDate('');
        setSelectedColor('#ffffff');
    };

    const toggleTask = (index) => {
        const newTasks = [...tasks];
        newTasks[index].completed = !newTasks[index].completed;
        setTasks(newTasks);
    };

    const deleteTask = (index) => {
        setTasks(tasks.filter((_, i) => i !== index));
    };

    const filteredTasks = tasks.filter(task => task.text.toLowerCase().includes(searchInput.toLowerCase()));

    return (
        <View style={styles.container}>
            <TextInput
                style={styles.input}
                placeholder="新しいタスクを追加"
                value={taskInput}
                onChangeText={setTaskInput}
            />
            <TextInput
                style={styles.input}
                placeholder="メモを追加"
                value={noteInput}
                onChangeText={setNoteInput}
            />
            <TextInput
                style={styles.input}
                placeholder="予定日 (YYYY-MM-DD)"
                value={selectedDate}
                onChangeText={setSelectedDate}
            />
            
            <ColorPicker
                style={{ height: 200 }}
                onColorSelected={color => setSelectedColor(color)}
                hideSliders={true}
            />
            <Button title="追加" onPress={addTask} />

            <TextInput
                style={styles.input}
                placeholder="タスクを検索"
                value={searchInput}
                onChangeText={setSearchInput}
            />

            <Calendar
                onDayPress={(day) => setSelectedDate(day.dateString)}
                markedDates={{
                    [selectedDate]: { selected: true, marked: true, selectedColor: selectedColor }
                }}
            />

            <FlatList
                data={filteredTasks}
                keyExtractor={(item, index) => index.toString()}
                renderItem={({ item, index }) => (
                    <View style={[styles.taskContainer, { backgroundColor: item.color }]}>
                        <Checkbox
                            status={item.completed ? 'checked' : 'unchecked'}
